    %% Phase 3: ECMP Traceroute Mapping
    subgraph P3[" 3. ECMP MULTI-PATH TRACEROUTE (mapper/ & internal/traceroute/) "]
        direction TB
        MapperAdd["Add Peer to Active Probing Queue
Mapper.AddPeer(node)"]:::mapperStyle
        PortSweepLoop["Iterate UDP Source Ports
(e.g., 33436, 33437 ... 33446)"]:::mapperStyle
        RawSocket["Construct Raw UDP Socket
syscall.SetsockoptInt(IP_TTL)
syscall.SetsockoptInt(IP_RECVERR)"]:::mapperStyle
        SendProbe["Transmit Probe Packet
Decrement TTL at each router hop"]:::mapperStyle
        EpollWait["Epoll Async Listener on Error Queue
syscall.Recvmsg(MSG_ERRQUEUE)"]:::mapperStyle
        CatchICMP["Capture ICMP Time Exceeded
Extract Hop IP (Spine/Leaf Switch)"]:::mapperStyle
        MergeGraph["Register Node & Links in NetworkGraph
graph.IncrRoute(path)"]:::mapperStyle

        PeerEvent --> MapperAdd --> PortSweepLoop --> RawSocket --> SendProbe --> EpollWait --> CatchICMP --> MergeGraph
    end

    %% Phase 4: High-Frequency UDP Ping Transport
    subgraph P4[" 4. CONTINUOUS UDP PING TRANSPORT (transport.go & pinger.go) "]
        direction TB
        PingerLoop["Continuous Sweep Loop (100ms Interval)
pinger.PingPeers(ctx)"]:::pingStyle
        MsgPackBuild["Serialize Packet via MessagePack
msgp.Encode(PingPacket{TimestampNS})"]:::pingStyle
        UDPSend["Send UDP Probe on Port 33435

        PeerAck["Remote Peer Transport Echoes ACK
transport.Listen() Reply"]:::pingStyle
        CalcRTT["Calculate Microsecond RTT
rtt = time.Since(start)"]:::pingStyle
        RingBuffer["Update 100-Sample Rolling Ring Buffer
Calculate Median, P90, P99 & Loss Rate"]:::pingStyle

        MergeGraph --> PingerLoop --> MsgPackBuild --> UDPSend --> PeerAck --> CalcRTT --> RingBuffer
    end
